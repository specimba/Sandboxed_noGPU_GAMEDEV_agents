   171→  /* ================= public API (used by React UI) ================= */
   172→
   173→  begin(): void {
   174→    if (this.phase !== 'title') return;
   175→    this.audio.unlock();

   179→    useGameStore.getState().set({ phase: 'flying' });

   182→
   183→  pause(): void {
   184→    if (this.phase !== 'playing') return;
   185→    this.phase = 'paused';
   186→    useGameStore.getState().set({ phase: 'paused' });
   187→  }
   188→
   189→  resume(): void {
   190→    if (this.phase !== 'paused') return;
   191→    this.audio.unlock();
   192→    this.phase = 'playing';
   193→    useGameStore.getState().set({ phase: 'playing' });
   194→  }
   195→
   196→  togglePause(): void {
   197→    if (this.phase === 'playing') this.pause();
   198→    else if (this.phase === 'paused') this.resume();
   199→  }

   201→  restartDepth(): void {
   202→    if (this.phase !== 'paused') return;

   205→    this.player.respawn(this.level!.spawn);
   206→    this.fadeTarget = 0;
   207→    this.phase = 'playing';
   208→    useGameStore.getState().set({ phase: 'playing', shards: 0, hearts: GAME.heartsMax, gateActive: false, gateDir: null });
   209→  }

   212→    this.clearLevelObjects();
   213→    this.depth = 1;
   214→    this.buildLevelObjects(1, (Math.random() * 100000) | 0);
   215→    this.attractAngle = Math.random() * Math.PI * 2;
   216→    this.fadeTarget = 0;
   217→    this.phase = 'title';
   218→    useGameStore.getState().set({
   219→      phase: 'title',
   220→      depth: 1,
   221→      shards: 0,
   222→      hearts: GAME.heartsMax,
   223→      gateActive: false,
   224→      gateDir: null,
   225→    });
   226→  }
   227→
   228→  respawn(): void {
   229→    if (this.phase !== 'dead') return;
   230→    this.player.respawn(this.level!.spawn);
   231→    for (const h of this.hunters) h.reset();
   232→    this.fadeTarget = 0;
   233→    this.phase = 'playing';
   234→    useGameStore.getState().set({ phase: 'playing', hearts: GAME.heartsMax, alert: 0 });
   235→  }
   236→
   237→  descend(): void {
   238→    if (this.phase !== 'cleared') return;
   239→    this.depth += 1;
   240→    const best = Math.max(this.depth, useGameStore.getState().bestDepth);
   241→    saveBestDepth(best);
   242→    this.clearLevelObjects();
   243→    this.buildLevelObjects(this.depth, (Math.random() * 100000) | 0);
   244→    this.player.hearts = Math.min(GAME.heartsMax, this.player.hearts + 1);
   245→    this.player.reposition(this.level!.spawn);
   246→    this.audio.setDepthMood(this.depth);
   247→    this.fadeTarget = 0;
   248→    this.phase = 'playing';
   249→    this.runStart = this.t;
   250→    useGameStore.getState().set({
   251→      phase: 'playing',
   252→      depth: this.depth,
   253→      bestDepth: best,
   254→      shards: 0,
   255→      hearts: this.player.hearts,
   256→      gateActive: false,
   257→      gateDir: null,
   258→    });
   259→    useGameStore
   260→      .getState()
   261→      .pushToast(`DEPTH ${depthRoman(this.depth)} — ${depthMeters(this.depth)}m BELOW`, 'gold');
   262→  }
   263→
   264→  toggleMute(): void {
   265→    const m = !useGameStore.getState().muted;
   266→    useGameStore.getState().set({ muted: m });

   277→  /* ================= level lifecycle ================= */
   278→
   279→  private buildLevelObjects(depth: number, seed: number): void {
   280→    const level = buildLevel(depth, seed, this.pulseUniform!);
   281→    this.level = level;
   282→    this.scene.add(level.group);
   283→
   284→    for (const p of level.shardPositions) this.shards.push(new Shard(this.scene, p));
   285→    this.gate = new Gate(this.scene, level.gatePosition, level.gateYaw);
   286→    for (const h of level.hunterHomes) this.hunters.push(new Hunter(this.scene, h, depth));
   287→
   288→    this.player.respawn(level.spawn);
   289→    // face down the path
   290→    this.rig.yaw = Math.atan2(
   291→      level.gatePosition.x - level.spawn.x,
   292→      level.gatePosition.z - level.spawn.z,
   293→    ) + Math.PI;
   294→    this.rig.pitch = 0.38;
   295→  }
   296→
   297→  private clearLevelObjects(): void {
   298→    for (const s of this.shards) s.dispose(this.scene);
   299→    this.shards = [];
   300→    for (const h of this.hunters) h.dispose(this.scene);
   301→    this.hunters = [];
   302→    this.gate?.dispose(this.scene);
   303→    this.gate = null;
   304→    this.level?.dispose();
   305→    this.level = null;
   306→    this.pending = [];
   307→    for (const w of this.waves) {
   308→      w.mesh.removeFromParent();
   309→      w.mat.dispose();
   310→      w.mesh.geometry.dispose();
   311→    }
   312→    this.waves = [];
   313→  }

   315→  /* ================= pulse system ================= */
   316→
   317→  private pulseIdx = 0;
   318→
   319→  private emitPulse(origin: THREE.Vector3, kind: 'teal' | 'gold' = 'teal', silent = false): void {
   320→    const pulses = this.pulseUniform!.value;
   321→    // round-robin slot so simultaneous echoes coexist in the shader buffer
   322→    this.pulseIdx = (this.pulseIdx + 1) % pulses.length;
   323→    pulses[this.pulseIdx].set(origin.x, origin.y, origin.z, this.t);
   324→
   325→    // visual shockwave sphere
   326→    const mat = createWaveMaterial();
   327→    mat.uniforms.uColor.value = kind === 'gold' ? COLORS.gate.clone() : new THREE.Color('#6ff2df');
   328→    const mesh = new THREE.Mesh(waveGeo, mat);
   329→    mesh.position.copy(origin);
   330→    mesh.frustumCulled = false;
   331→    mesh.renderOrder = 8;
   332→    this.scene.add(mesh);
   333→    this.waves.push({ mesh, mat, t0: this.t, origin: origin.clone() });
   334→
   335→    if (!silent) {
   336→      this.audio.pulse();
   337→      this.rig.addFovKick(5);
   338→      this.rig.addShake(0.1);
   339→    }
   340→
   341→    // schedule wavefront hits at physically-correct delays
   342→    const schedule = (obj: THREE.Vector3, fn: () => void) => {
   343→      const d = obj.distanceTo(origin);
   344→      if (d > WAVE.revealRadius) return;
   345→      this.pending.push({ t: this.t + d / WAVE.speed, fn });
   346→    };
   347→    for (const s of this.shards) {
   348→      if (s.taken) continue;
   349→      const i = this.shards.indexOf(s);
   350→      schedule(
   351→        s.group.position,
   352→        () => {
   353→          s.bloom();
   354→          this.audio.chime(0, 1100 + i * 140, 0.14);
   355→        },
   356→      );
   357→    }
   358→    if (this.gate && !this.gate.active) {
   359→      schedule(this.gate.group.position, () => {
   360→        this.gate!.reveal = 1;
   361→        this.audio.chime(0, 480, 0.12);
   362→      });
   363→    }
   364→    for (const h of this.hunters) {
   365→      const d = h.pos.distanceTo(origin);
   366→      if (d > WAVE.revealRadius) continue;
   367→      if (d < WAVE.stunRadius) {
   368→        this.pending.push({
   369→          t: this.t + d / WAVE.speed,
   370→          fn: () => {
   371→            h.onPulse(origin, 'stun');
   372→            this.audio.stun();
   373→            this.particles.burst(h.pos, 14, 4, { color: COLORS.hazard, life: 0.7, size: 0.4 });
   374→          },
   375→        });
   376→      } else if (d < WAVE.alertRadius) {
   377→        this.pending.push({
   378→          t: this.t + d / WAVE.speed,
   379→          fn: () => {
   380→            h.onPulse(origin, 'alert');
   381→            this.audio.growl(0);
   382→          },
   383→        });
   384→      } else {
   385→        this.pending.push({ t: this.t + d / WAVE.speed, fn: () => h.onPulse(origin, 'reveal') });
   386→      }
   387→    }
   388→
   389→    if (kind === 'gold') {
   390→      this.particles.burst(origin, 40, 7, { color: COLORS.gate, life: 1.2, size: 0.5 });
   391→    }
   392→  }

   394→  /* ================= damage / death ================= */
   395→
   396→  private damage(from: THREE.Vector3, scatterHunters: boolean): void {
   397→    if (this.player.invuln > 0 || this.phase !== 'playing') return;
   398→    this.player.hearts -= 1;
   399→    const store = useGameStore.getState();
   400→    store.set({ hearts: Math.max(0, this.player.hearts) });
   401→    this.vignette.uniforms.uHurt.value = 1;
   402→    this.rig.addShake(0.55);
   403→    this.audio.hit();
   404→    const away = this.player.pos.clone().sub(from).setY(0);
   405→    if (away.lengthSq() < 0.01) away.set(0, 0, 1);
   406→    away.normalize().multiplyScalar(13);
   407→    this.player.vel.set(away.x, 6.5, away.z);
   408→    this.player.invuln = 1.4;
   409→    if (scatterHunters) for (const h of this.hunters) h.scatter();
   410→    if (this.player.hearts <= 0) this.die();
   411→  }

   413→  private die(): void {
   414→    this.phase = 'dead';
   415→    this.deadAt = this.t;
   416→    this.fadeTarget = 0.9;
   417→    this.audio.death();
   418→    this.audio.setAlert(0);
   419→    const store = useGameStore.getState();
   420→    store.set({ phase: 'dead', deaths: store.deaths + 1, alert: 0 });
   421→    this.particles.burst(this.player.pos, 50, 9, { color: COLORS.player, life: 1.2, size: 0.5 });
   422→  }

   424→  /* ================= main loop ================= */
   425→
   426→  private loop = () => {
   427→    if (this.disposed) return;
   428→    this.raf = requestAnimationFrame(this.loop);
   429→    const dt = Math.min(0.05, this.clock.getDelta());
   430→    this.t += dt;
   431→
   432→    this.frameEma = this.frameEma * 0.95 + dt * 1000 * 0.05;
   433→    this.adaptDpr(dt);
   434→
   435→    switch (this.phase) {
   436→      case 'title':
   437→        this.updateTitle(dt);
   438→        break;
   439→      case 'flying':
   440→        this.updateFlying(dt);
   441→        break;
   442→      case 'playing':
   443→        this.updatePlaying(dt);
   444→        break;
   445→      case 'paused':
   446→      case 'dead':
   447→      case 'cleared':
   448→        this.updateFrozen(dt);
   449→        break;
   450→    }
   451→
   452→    // shared uniforms
   453→    for (const u of [this.level?.material.uniforms, this.dust.uniforms]) {
   454→      if (!u) continue;
   455→      if ('uTime' in u) u.uTime.value = this.t;
   456→      if ('uPlayerPos' in u) (u.uPlayerPos.value as THREE.Vector3).copy(this.player.pos);
   457→    }
   458→    this.dust.uniforms.uPix.value = this.dpr;
   459→
   460→    // vignette state
   461→    const v = this.vignette.uniforms;
   462→    v.uTime.value = this.t;
   463→    v.uHurt.value = Math.max(0, v.uHurt.value - dt * 1.6);
   464→    v.uAlert.value += (this.alertLevel - v.uAlert.value) * Math.min(1, dt * 3);
   465→    v.uFade.value += (this.fadeTarget - v.uFade.value) * Math.min(1, dt * 2.2);
   466→
   467→    this.composer.render();
   468→    this.input.clearFrame();
   469→  };
   470→
   471→  private updateTitle(dt: number): void {
   472→    this.attractAngle += dt * 0.06;
   473→    const center = this.level!.center;
   474→    this.rig.setOrbit(center, this.attractAngle, 26, 15);
   475→    this.camera.position.copy(this.rig.currentPos);
   476→    this.camera.lookAt(this.rig.currentLook);
   477→
   478→    this.attractPulse -= dt;
   479→    if (this.attractPulse <= 0) {
   480→      this.attractPulse = 3.4;
   481→      const n = this.level!.colliders[(Math.random() * this.level!.colliders.length) | 0];
   482→      this.emitPulse(new THREE.Vector3(n.x, n.top + 0.5, n.z), 'teal', true);
   483→    }
   484→    this.updateAmbient(dt);
   485→    if (this.input.consumeBegin()) this.begin();
   486→  }
   487→
   488→  private updateFlying(dt: number): void {
   489→    // camera flies to the wisp; world idles
   490→    this.updateAmbient(dt);
   491→    this.rig.update(dt, this.camera, this.player.pos, ZERO, this.level!.colliders, this.reduceFx);
   492→    this.player.group.position.copy(this.player.pos);
   493→    this.player.group.position.y += Math.sin(this.t * 2) * 0.06;
   494→    if (!this.rig.flying) {
   495→      this.phase = 'playing';
   496→      this.runStart = this.t;
   497→      const store = useGameStore.getState();
   498→      store.set({ phase: 'playing', runTime: 0 });
   499→      store.pushToast(`DEPTH ${depthRoman(this.depth)} — ${depthMeters(this.depth)}m BELOW`, 'gold');
   500→      store.pushToast('CLICK / F — ECHO PULSE', 'info');
   501→    }
   502→  }
   503→
   504→  private updatePlaying(dt: number): void {
   505→    const store = useGameStore.getState();
   506→    const level = this.level!;
   507→
   508→    // --- input actions ---
   509→    if (this.input.consumePause()) {
   510→      this.pause();
   511→      return;
   512→    }
   513→    if (this.input.consumePulse()) this.tryPulse();
   514→    if (this.input.consumeJump()) this.player.requestJump();
   515→    if (this.input.consumeDash()) {
   516→      const yaw = this.rig.yaw;
   517→      const mx = this.input.moveX;
   518→      const my = this.input.moveY;
   519→      const fwdX = -Math.sin(yaw);
   520→      const fwdZ = -Math.cos(yaw);
   521→      const dx = fwdX * my + -fwdZ * mx;
   522→      const dz = fwdZ * my + fwdX * mx;
   523→      if (this.player.tryDash(dx, dz)) this.audio.dash();
   524→    }
   525→
   526→    // camera orbit
   527→    this.rig.orbit(this.input.lookX, this.input.lookY);
   528→    this.rig.rotate(this.input.rotateAxis, dt);
   529→    if (this.input.zoom !== 0) this.rig.zoomBy(this.input.zoom);
   530→
   531→    // --- player physics ---
   532→    this.player.update(
   533→      dt,
   534→      { moveX: this.input.moveX, moveY: this.input.moveY, jumpHeld: this.input.jumpHeld },
   535→      this.rig.yaw,
   536→      level.colliders,
   537→      this.playerEvents,
   538→    );
   539→    if (this.playerEvents.landed) {
   540→      this.audio.land();
   541→      this.particles.burst(this.player.pos, 8, 2.5, { color: COLORS.rock, life: 0.4, size: 0.3 });
   542→    }
   543→    if (this.playerEvents.jumped) this.audio.jump();
   544→
   545→    // fell into the abyss
   546→    if (this.player.pos.y < PLAYER.killY) {
   547→      this.damage(this.player.pos, false);
   548→      if (this.player.hearts > 0) this.player.reposition(level.spawn);
   549→    }
   550→
   551→    // --- spikes ---
   552→    for (const s of level.spikes) {
   553→      if (Math.hypot(this.player.pos.x - s.x, this.player.pos.z - s.z) < s.r + 0.3) {
   554→        if (this.player.pos.y < s.top + 1.2) {
   555→          const spikePos = new THREE.Vector3(s.x, this.player.pos.y - 0.5, s.z);
   556→          this.damage(spikePos, false);
   557→        }
   558→      }
   559→    }
   560→
   561→    // --- shards ---
   562→    let collected = -1;
   563→    for (let i = 0; i < this.shards.length; i++) {
   564→      const s = this.shards[i];
   565→      s.update(dt, this.t);
   566→      if (!s.taken && s.group.position.distanceTo(this.player.pos) < 1.5) collected = i;
   567→    }
   568→    if (collected >= 0) {
   569→      this.shards[collected].collect(this.particles);
   570→      this.audio.shard(collected);
   571→      const count = this.shards.filter((s) => s.taken).length;
   572→      store.set({ shards: count });
   573→      store.pushToast(`ECHO SHARD ${count}/${GAME.shardsNeeded}`, 'gold');
   574→      if (count >= GAME.shardsNeeded && this.gate && !this.gate.active) {
   575→        this.gate.activate(this.particles);
   576→        this.audio.gateOpen();
   577→        store.set({ gateActive: true });
   578→        store.pushToast('THE GATE AWAKENS — REACH THE GOLD LIGHT', 'gold');
   579→        this.emitPulse(this.gate.group.position.clone().add(new THREE.Vector3(0, 2, 0)), 'gold');
   580→      }
   581→    }
   582→
   583→    // --- gate ---
   584→    this.gate?.update(dt, this.t);
   585→    if (this.gate?.active) {
   586→      const d = Math.hypot(
   587→        this.player.pos.x - this.gate.group.position.x,
   588→        this.player.pos.z - this.gate.group.position.z,
   589→      );
   590→      if (d < 2.2 && Math.abs(this.player.pos.y - this.gate.group.position.y) < 3) {
   591→        this.clearDepth();
   592→        return;
   593→      }
   594→    }
   595→
   596→    // --- hunters ---
   597→    let alert = 0;
   598→    for (const h of this.hunters) {
   599→      const hit = h.update(dt, this.t, this.player.pos, 1);
   600→      if (hit && this.player.invuln <= 0) this.damage(h.pos, true);
   601→      if (h.state === 'alert') {
   602→        const prox = 1 - Math.min(1, h.pos.distanceTo(this.player.pos) / 30);
   603→        alert = Math.max(alert, prox);
   604→      }
   605→    }
   606→    this.alertLevel = alert;
   607→    this.audio.setAlert(alert);
   608→
   609→    // --- fx ---
   610→    this.updateAmbient(dt);
   611→
   612→    // --- camera ---
   613→    this.rig.update(dt, this.camera, this.player.pos, this.player.vel, level.colliders, this.reduceFx);
   614→
   615→    // --- HUD vitals (throttled) ---
   616→    this.vitalsTimer -= dt;
   617→    if (this.vitalsTimer <= 0) {
   618→      this.vitalsTimer = 0.09;
   619→      const st = useGameStore.getState();
   620→      st.set({
   621→        pulseReady: 1 - Math.min(1, this.pulseCd / WAVE.cooldown),
   622→        dashReady: this.player.dashReady ? 1 : 0,
   623→        alert,
   624→        runTime: this.t - this.runStart,
   625→        gateDir: this.gate?.active ? this.gateAngle() : null,
   626→      });
   627→    }
   628→  }

   630→  private pulseCd = 0;

   632→  private tryPulse(): void {
   633→    if (this.pulseCd > 0) return;
   634→    this.pulseCd = WAVE.cooldown;

   638→  private gateAngle(): number {
   639→    if (!this.gate) return 0;
   640→    const dx = this.gate.group.position.x - this.player.pos.x;
   641→    const dz = this.gate.group.position.z - this.player.pos.z;
   642→    return Math.atan2(dx, dz) + this.rig.yaw;
   643→  }

   674→    this.gate?.update(dt, this.t);

   677→  private updateWaves(): void {
   678→    const speed = WAVE.speed;
   679→    for (let i = this.waves.length - 1; i >= 0; i--) {
   680→      const w = this.waves[i];
   681→      const age = this.t - w.t0;
   682→      const radius = age * speed;
   683→      if (radius > 110 || age > 4.2) {
   684→        w.mesh.removeFromParent();
   685→        w.mat.dispose();
   686→        this.waves.splice(i, 1);
   687→        continue;
   688→      }
   689→      // the shell is a launch cue only — hand over to the geometry band
   690→      w.mesh.visible = radius < 26;
   691→      w.mesh.scale.setScalar(Math.max(0.01, radius));
   692→      w.mat.uniforms.uAge.value = age;
   693→    }
   694→  }

   698→  private adaptDpr(dt: number): void {
   699→    this.dprCheck += dt;
   700→    if (this.dprCheck < 2) return;
   701→    this.dprCheck = 0;
   702→    const target = Math.min(window.devicePixelRatio || 1, 2);
   703→    if (this.frameEma > 21 && this.dpr > 1) {
   704→      this.dpr = Math.max(1, this.dpr - 0.25);
   705→      this.applyDpr();
   706→    } else if (this.frameEma < 13.5 && this.dpr < target) {
   707→      this.dpr = Math.min(target, this.dpr + 0.25);
   708→      this.applyDpr();
   709→    }
   710→  }

   713→    this.renderer.setPixelRatio(this.dpr);
   714→    this.composer.setPixelRatio(this.dpr);
   715→    this.particles.setPixelRatio(this.dpr);
   716→    this.resize();
   717→  }

   720→    const w = this.canvas.clientWidth || window.innerWidth;

   723→    this.composer.setSize(w, h);
   724→    this.camera.aspect = w / h;
   725→    this.camera.updateProjectionMatrix();
   726→    this.dpr = this.renderer.getPixelRatio();
   727→  }
